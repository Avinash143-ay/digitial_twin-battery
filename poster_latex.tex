% ---------------- Problem Statement ----------------
\begin{block}{\Large\textbf{🎯 Problem Statement}}
\large
\textbf{Challenge:} Battery management systems need accurate, real-time voltage and temperature prediction with uncertainty quantification for:

\begin{itemize}
    \item ⚡ Safety monitoring (thermal runaway prevention)
    \item 🔋 State-of-Health (SOH) estimation
    \item 📊 Remaining useful life prediction
    \item ⚙️ Adaptive charging strategies
\end{itemize}

\vspace{0.5cm}
\textbf{Gap:} Traditional physics-based models are slow; single AI models don't capture uncertainty; need multiple approaches for validation.
\end{block}

\vspace{1cm}

% ---------------- Our Solution ----------------
\begin{alertblock}{\Large\textbf{💡 Our Solution}}
\large
\textbf{Dual-Model Architecture} combining:

\begin{enumerate}
    \item \textbf{Transformer Model} - Fast parallel predictions (<50ms)
    \item \textbf{Deep Ensemble Model} - Uncertainty quantification (10 models)
\end{enumerate}

\vspace{0.5cm}
\textbf{Key Innovation:} MoE-Enhanced Transformer with 1000+ experts achieving \textbf{0.158\% voltage prediction error} on challenging 0°C KIT battery aging data (P065 cell: 0°C, 10-100\% SOC).
\end{alertblock}

\vspace{1cm}

% ---------------- System Architecture ----------------
\begin{block}{\Large\textbf{🏗️ System Architecture}}
\large
\textbf{Why Two Models on Same Dataset?}
\begin{itemize}
    \item Validation: Cross-verification of predictions
    \item Speed vs Uncertainty Tradeoff
    \item Different Capabilities
    \item Research Question: Which AI is better?
\end{itemize}

\vspace{0.5cm}

% ---------------- Transformer Model ----------------
\begin{block}{\Large\textbf{🧠 Model 1: MoE-Enhanced Transformer}}
\large
\textbf{Architecture:}
\begin{itemize}
    \item \textbf{Input:} [Relative\_Age, V₀, T₀, Current Sequence]
    \item \textbf{Encoder:} 2-layer Transformer (2 heads/layer)
    \item \textbf{MoE Layers:} 3 cascaded (1000+ experts, top-20)
    \item \textbf{Output:} V(μ,σ), T(μ,σ)
    \item \textbf{Max Horizon:} 150 steps
\end{itemize}

\vspace{0.3cm}
\textbf{Key Features:}
\begin{itemize}
    \item ✓ Mixture of Experts (1000+ specialists)
    \item ✓ Top-20 dynamic routing
    \item ✓ Uncertainty quantification (μ, σ)
    \item ✓ \textbf{0.158\% voltage MAPE} ⭐
    \item ✓ Trained on KIT P065 (0°C)
\end{itemize}

\vspace{0.3cm}
\textbf{Preprocessing:}
\begin{itemize}
    \item Voltage: V/3
    \item Temperature: T/30
    \item Current: I/5
\end{itemize}
\end{block}

\vspace{1cm}

% ---------------- Deep Ensemble Model ----------------
\begin{block}{\Large\textbf{🔬 Model 2: Deep Ensemble (Uncertainty)}}
\large
\textbf{Architecture:}
\begin{itemize}
    \item \textbf{Input:} [Relative\_Age, V, T, I₀, I\_next]
    \item \textbf{Ensemble Size:} 10 independent networks
    \item \textbf{Parameters:} 200K each → 2M total
    \item \textbf{Activation:} GELU (smooth gradients)
    \item \textbf{Max Horizon:} 75 steps (2-sec intervals)
\end{itemize}

\vspace{0.3cm}
\textbf{Key Features:}
\begin{itemize}
    \item ✓ Autoregressive prediction
    \item ✓ Uncertainty quantification (spread)
    \item ✓ Trained on KIT P065 (0°C - same as Transformer)
    \item ✓ Handles aging effects
    \item ✓ Robust to measurement noise
\end{itemize}

\vspace{0.3cm}
\textbf{Prediction Strategy:}
\begin{itemize}
    \item Each model predicts independently
    \item Median of 10 predictions = final output
    \item Spread indicates confidence
\end{itemize}
\end{block}

\vspace{1cm}

% ---------------- Dataset Info ----------------
\begin{alertblock}{\Large\textbf{📊 Dataset}}
\large
\textbf{KIT Battery Aging Dataset}

\begin{itemize}
    \item \textbf{Source:} Nature Scientific Data 2024
    \item \textbf{Cell:} P065 (18650 Li-ion, 2Ah)
    \item \textbf{Conditions:} 0°C temperature, 10-100\% SOC
    \item \textbf{Usage:} SAME cell data for both models
    \item \textbf{Difference:} AI architectures
    \begin{itemize}
        \item Transformer: Attention-based, uses Relative\_Age
        \item Ensemble: Voting-based, uses Relative\_Age
    \end{itemize}
\end{itemize}
\end{alertblock}
